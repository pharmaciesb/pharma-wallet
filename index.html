// On ajoute la bibliothèque de lecture de QR Code const script = document.createElement('script'); script.src = "https://cdn.jsdelivr.net/npm/jsqr@1.4.0/dist/jsQR.min.js"; document.head.appendChild(script); async function decodeDisplayedQR() { addLog("Tentative de lecture visuelle du QR Code..."); try { // 1. On demande l'autorisation de capturer l'onglet const stream = await navigator.mediaDevices.getDisplayMedia({ video: { displaySurface: "browser" } }); const track = stream.getVideoTracks()[0]; const imageCapture = new ImageCapture(track); const bitmap = await imageCapture.grabFrame(); // 2. On dessine l'image sur un canvas invisible pour l'analyser const canvas = document.createElement('canvas'); canvas.width = bitmap.width; canvas.height = bitmap.height; const ctx = canvas.getContext('2d'); ctx.drawImage(bitmap, 0, 0); const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height); // 3. Utilisation de jsQR pour trouver le code dans l'image const code = jsQR(imageData.data, imageData.width, imageData.height); if (code) { addLog("QR Code décodé visuellement !", "success"); const urlFinal = code.data; addLog("URL extraite : " + urlFinal); // Souvent le HCN est à la fin de cette URL const hcnMatch = urlFinal.match(/HCN[a-zA-Z0-9]+/); if (hcnMatch) { showBarcode(hcnMatch[0]); } else { addLog("URL trouvée mais pas de HCN dedans. Analyse manuelle requise.", "error"); } } else { addLog("Aucun QR Code détecté sur l'écran.", "error"); } track.stop(); // On arrête la capture } catch (err) { addLog("Erreur de capture : " + err.message, "error"); } }
